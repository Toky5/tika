import org.junit.jupiter.api.extension.ExtendWith;

public class EndianUtils_readIntBE_7_0_Test {

    @Test
    public void testReadIntBE() throws IOException, BufferUnderrunException {
        // Test with a valid input
        InputStream stream = new ByteArrayInputStream(new byte[] { 0, 0, 0, 1 });
        int result = EndianUtils.readIntBE(stream);
        assertEquals(1, result);
        // Test with a negative input
        stream = new ByteArrayInputStream(new byte[] { 0, 0, 0, -1 });
        result = EndianUtils.readIntBE(stream);
        assertEquals(-1, result);
        // Test with a buffer underrun
        stream = new ByteArrayInputStream(new byte[] { 0, 0, 0 });
        assertThrows(BufferUnderrunException.class, () -> {
            EndianUtils.readIntBE(stream);
        });
    }
}
