public class EndianUtils_readUShortBE_3_0_Test {

    @Test
    public void testReadUShortBE() throws IOException, BufferUnderrunException {
        // Test with valid input
        InputStream stream = new ByteArrayInputStream(new byte[] { 0x12, 0x34 });
        int result = EndianUtils.readUShortBE(stream);
        assertEquals(0x1234, result);
        // Test with empty input
        stream = new ByteArrayInputStream(new byte[] {});
        assertThrows(BufferUnderrunException.class, () -> EndianUtils.readUShortBE(stream));
        // Test with null input
        stream = null;
        assertThrows(NullPointerException.class, () -> EndianUtils.readUShortBE(stream));
    }
}
