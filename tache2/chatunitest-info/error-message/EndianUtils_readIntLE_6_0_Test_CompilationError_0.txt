class EndianUtils_readIntLE_6_0_Test {

    @Test
    void testReadIntLE() throws IOException, BufferUnderrunException {
        // Test with a valid input
        byte[] input = new byte[] { 0x01, 0x02, 0x03, 0x04 };
        InputStream stream = new ByteArrayInputStream(input);
        int result = EndianUtils.readIntLE(stream);
        assertEquals(0x04030201, result);
        // Test with a buffer underrun
        input = new byte[] { 0x01, 0x02, 0x03 };
        stream = new ByteArrayInputStream(input);
        assertThrows(BufferUnderrunException.class, () -> EndianUtils.readIntLE(stream));
    }
}
