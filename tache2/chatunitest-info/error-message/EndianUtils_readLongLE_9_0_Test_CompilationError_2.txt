package org.apache.tika.io;

import org.apache.tika.io.EndianUtils;
import java.io.ByteArrayInputStream;
import java.io.IOException;
import java.io.InputStream;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import org.apache.tika.exception.TikaException;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

public class EndianUtils_readLongLE_9_0_Test {

    @Test
    public void testReadLongLE() throws IOException, BufferUnderrunException {
        // Test with a valid long value
        long value = 1234567890123456789L;
        byte[] bytes = new byte[8];
        for (int i = 0; i < 8; i++) {
            bytes[i] = (byte) (value >> (i * 8));
        }
        InputStream stream = new ByteArrayInputStream(bytes);
        long result = EndianUtils.readLongLE(stream);
        assertEquals(value, result);
        // Test with a negative long value
        value = -1234567890123456789L;
        bytes = new byte[8];
        for (int i = 0; i < 8; i++) {
            bytes[i] = (byte) (value >> (i * 8));
        }
        stream = new ByteArrayInputStream(bytes);
        result = EndianUtils.readLongLE(stream);
        assertEquals(value, result);
        // Test with a buffer underrun
        stream = new ByteArrayInputStream(new byte[7]);
        assertThrows(BufferUnderrunException.class, () -> EndianUtils.readLongLE(stream));
    }

    public static class BufferUnderrunException extends IOException {

        public BufferUnderrunException() {
            super("Buffer underrun");
        }
    }
}
